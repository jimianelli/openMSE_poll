Data fitting walkthrough and MSE demo

Overview

This document converts openMSE/R/sca_openmse.R into a narrative Quarto report. It walks through how the pm_24 data are parsed and mapped to the SCA model inputs, documents key parameter choices, and produces figures and tables that summarize the fit and downstream MSE outputs.

From a scientific perspective, the workflow links observed removals and survey indices to an age-structured assessment, derives life-history relationships, and propagates those outputs into an operating model and management strategy evaluation. Key run settings are summarized in Table 1, the temporal and age coverage in Table 2, and the primary input signals are visualized in Figure 1 and Figure 2. Growth and maturity derivations are documented in Table 3 and Figure 3, Figure 4, and Figure 5, while the assessment and MSE outputs are synthesized in Table 5, Figure 6, and ?@fig-mse-metrics-plot, with MP comparisons in Figure 8, Figure 9, and Figure 10 and the quilt summary in Table 9.

# Data parser for ADMB-format tab-delimited .dat files
read_data <- function(file) {
  lines <- readLines(file)
  result <- list()
  current_name <- NULL
  buffer <- list()

  for (line in lines) {
    line <- trimws(line)
    if (line == "") next
    if (startsWith(line, "#")) {
      if (!is.null(current_name)) {
        result[[current_name]] <- if (length(buffer) == 1) buffer[[1]] else do.call(rbind, buffer)
      }
      current_name <- sub("^#", "", line)
      buffer <- list()
    } else {
      nums <- as.numeric(unlist(strsplit(line, "\\s+")))
      buffer[[length(buffer) + 1]] <- nums
    }
  }

  if (!is.null(current_name)) {
    result[[current_name]] <- if (length(buffer) == 1) buffer[[1]] else do.call(rbind, buffer)
  }

  result
}

if (file.exists(dat_rds)) {
  dat <- readRDS(dat_rds)
  data_source <- "pm_24.rds"
} else {
  if (!file.exists(dat_file)) {
    stop("Missing data file: ", dat_file,
         "\nRun once with pm_24.dat present to create the cached RDS: ", dat_rds)
  }
  dat <- read_data(dat_file)
  saveRDS(dat, dat_rds)
  data_source <- "pm_24.dat (cached to RDS)"
}

Growth, weight-length, and maturity

We derive length-at-age from the Von Bertalanffy parameters and fit a weight-length relationship from the mean fishery weights-at-age. These life-history relationships determine the conversion between age, length, and biomass in both the assessment and the operating model. Derived parameters are summarized in Table 3, with supporting trajectories shown in Figure 3, Figure 4, and Figure 5.

Show code 
# Derive t0 from L0 (length at recage)
vbt0 <- recage + log(1 - L0 / vbLinf) / vbK

# Predicted length-at-age
len_at_age <- vbLinf * (1 - exp(-vbK * (ages - vbt0)))

# Mean weight-at-age from fishery data
mean_wt_age <- colMeans(wt_fsh)

# Fit weight-length: log(W) = log(a) + b * log(L)
wl_fit <- lm(log(mean_wt_age) ~ log(len_at_age))
wla <- exp(coef(wl_fit)[1])
wlb <- coef(wl_fit)[2]

# Derive L50 and L95 from maturity-at-age interpolated to length-at-age
L50 <- approx(p_mature, len_at_age, xout = 0.5)$y

Parameters in Table 1 and the data provenance in Table 4 are embedded here, while the life-history relationships shown in Figure 3 and Figure 5 ensure biological consistency between the assessment and OM conditioning stages.

pollock_data@AddIunits <- c(1L, 1L, 1L)

pollock_data@AddIndV <- array(1, dim = c(1, 3, max(ages) + 1))

# Catch-at-age: pad age-0 column with zeros

```{r}
# Data parser for ADMB-format tab-delimited .dat files
read_data <- function(file) {
  lines <- readLines(file)
  result <- list()
  current_name <- NULL
  buffer <- list()

  for (line in lines) {
    line <- trimws(line)
    if (line == "") next
    if (startsWith(line, "#")) {
      if (!is.null(current_name)) {
        result[[current_name]] <- if (length(buffer) == 1) buffer[[1]] else do.call(rbind, buffer)
      }
      current_name <- sub("^#", "", line)
      buffer <- list()
    } else {
      nums <- as.numeric(unlist(strsplit(line, "\\s+")))
      buffer[[length(buffer) + 1]] <- nums
    }
  }

  if (!is.null(current_name)) {
    result[[current_name]] <- if (length(buffer) == 1) buffer[[1]] else do.call(rbind, buffer)
  }

  result
}

if (file.exists(dat_rds)) {
  dat <- readRDS(dat_rds)
```

## Growth, weight-length, and maturity

We derive length-at-age from the Von Bertalanffy parameters and fit a
weight-length relationship from the mean fishery weights-at-age.
These life-history relationships determine the conversion between age,
length, and biomass in both the assessment and the operating model. Derived
parameters are summarized in @tbl-growth-params, with supporting
trajectories shown in @fig-length-at-age, @fig-weight-length, and
@fig-maturity-curve.

```{r}
#| label: tbl-growth-params
#| tbl-cap: "Derived growth and maturity parameters."

# Derive t0 from L0 (length at recage)
vbt0 <- recage + log(1 - L0 / vbLinf) / vbK

# Predicted length-at-age
len_at_age <- vbLinf * (1 - exp(-vbK * (ages - vbt0)))

# Mean weight-at-age from fishery data
mean_wt_age <- colMeans(wt_fsh)

# Fit weight-length: log(W) = log(a) + b * log(L)
wl_fit <- lm(log(mean_wt_age) ~ log(len_at_age))
wla <- exp(coef(wl_fit)[1])
wlb <- coef(wl_fit)[2]

# Derive L50 and L95 from maturity-at-age interpolated to length-at-age
L50 <- approx(p_mature, len_at_age, xout = 0.5)$y
L95 <- approx(p_mature, len_at_age, xout = 0.95)$y

growth_tbl <- data.frame(
  Parameter = c("vbt0", "wla", "wlb", "L50", "L95"),
  Value = c(round(vbt0, 3), signif(wla, 4), round(wlb, 4), round(L50, 1), round(L95, 1))
```

```{r}
# Maturity-at-age: replicate across years
mat_mat <- matrix(p_mature, nrow = nyears, ncol = nages, byrow = TRUE)

laa_mat <- matrix(len_at_age, nrow = nyears, ncol = nages, byrow = TRUE)

  arr1 <- array(t(m), dim = c(1, ncol(m), nrow(m)))
```

```{r}
      values <- array(NA_real_, dim = c(1, nrow(perf_tbl), length(metrics)))
      values[1, , ] <- as.matrix(perf_tbl[, metrics, drop = FALSE])

      quilt <- Slick::Quilt(
        Code = metrics,
        Label = metrics,
        Description = metrics,
        Value = values
      )
      Slick::Quilt(slick_obj) <- quilt
    }
  }

  Slick::Check(slick_obj)
  saveRDS(slick_obj, file.path(report_dir, "pollock_slick.rds"))
} else {
  cat("Package 'Slick' is not installed; skipping Slick object export.")
}
```
